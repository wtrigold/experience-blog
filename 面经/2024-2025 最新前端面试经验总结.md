### 2. 手写代码题（最高频）

#### 必背手写题 TOP 10
1. **Promise / Promise.all**
```js
function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      throw new TypeError('必须传入数组')
    }
    let results = []
    let count = 0
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        res => {
          results[i] = res
          count++
          if (count === promises.length) resolve(results)
        },
        err => reject(err)
      )
    })
  })
}
```

2. **深拷贝（考虑循环引用）**
```js
function deepClone(obj, hash = new WeakMap()) {
  if (obj === null || typeof obj !== 'object') return obj
  if (hash.has(obj)) return hash.get(obj)
  const clone = Array.isArray(obj) ? [] : {}
  hash.set(obj, clone)
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      clone[key] = deepClone(obj[key], hash)
    }
  }
  return clone
}
```

3. **防抖 & 节流**
```js
// 防抖
function debounce(fn, wait) {
  let timer = null
  return function(...args) {
    clearTimeout(timer)
    timer = setTimeout(() => fn.apply(this, args), wait)
  }
}

// 节流
function throttle(fn, delay) {
  let lastTime = 0
  return function(...args) {
    const now = Date.now()
    if (now - lastTime >= delay) {
      lastTime = now
      fn.apply(this, args)
    }
  }
}
```

4. **bind**
```js
Function.prototype.myBind = function(context, ...args1) {
  const fn = this
  function boundFn(...args2) {
    const isNew = this instanceof boundFn
    return fn.apply(isNew ? this : context, [...args1, ...args2])
  }
  boundFn.prototype = Object.create(fn.prototype)
  return boundFn
}
```

5. **数组拍平 flat**
```js
function flat(arr, depth = 1) {
  if (depth <= 0) return arr.slice()
  return arr.reduce((acc, val) => {
    return acc.concat(Array.isArray(val) ? flat(val, depth - 1) : val)
  }, [])
}
```

6. **数组转树**
```js
function arrayToTree(arr, parentId = 0) {
  const map = new Map()
  const tree = []
  arr.forEach(item => {
    map.set(item.id, { ...item, children: [] })
  })
  arr.forEach(item => {
    const node = map.get(item.id)
    if (item.pid === parentId) {
      tree.push(node)
    } else {
      const parent = map.get(item.pid)
      if (parent) parent.children.push(node)
    }
  })
  return tree
}
```

7. **发布订阅模式**
```js
class EventEmitter {
  constructor() {
    this.events = {}
  }
  on(event, fn) {
    if (!this.events[event]) this.events[event] = []
    this.events[event].push(fn)
  }
  emit(event, ...args) {
    if (this.events[event]) {
      this.events[event].forEach(fn => fn(...args))
    }
  }
  off(event, fn) {
    if (!this.events[event]) return
    this.events[event] = this.events[event].filter(f => f !== fn)
  }
}
```

```js
// 快速排序
function quickSort(arr) {
  if (arr.length <= 1) return arr
  const pivot = arr[Math.floor(arr.length / 2)]
  const left = arr.filter(x => x < pivot)
  const middle = arr.filter(x => x === pivot)
  const right = arr.filter(x => x > pivot)
  return [...quickSort(left), ...middle, ...quickSort(right)]
}

// 最大子数组和
function maxSubArray(nums) {
  let max = nums[0], sum = 0
  for (let num of nums) {
    sum = Math.max(num, sum + num)
    max = Math.max(max, sum)
  }
  return max
}
```

#### 实现一个带过期时间的 localStorage
```js
class Storage {
  set(key, value, expire) {
    const data = {
      value,
      expire: expire ? Date.now() + expire : null
    }
    localStorage.setItem(key, JSON.stringify(data))
  }
  
  get(key) {
    const str = localStorage.getItem(key)
    if (!str) return null
    const { value, expire } = JSON.parse(str)
    if (expire && Date.now() > expire) {
      localStorage.removeItem(key)
      return null
    }
    return value
  }
}
```

#### 并发控制调度器
```js
class Scheduler {
  constructor(limit) {
    this.limit = limit
    this.queue = []
    this.running = 0
  }
  
  add(task) {
    return new Promise(resolve => {
      const run = async () => {
        this.running++
        await task()
        this.running--
        if (this.queue.length) {
          this.queue.shift()()
        }
      }
      
      if (this.running < this.limit) {
        run().then(resolve)
      } else {
        this.queue.push(() => run().then(resolve))
      }
    })
  }
}
```
